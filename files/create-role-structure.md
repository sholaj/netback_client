#!/bin/bash

# create_ansible_role.sh
# This script creates a complete Ansible role structure for manage_server_list
# It creates all necessary directories and files with basic content

# Set role name
ROLE_NAME="manage_server_list"

# Create the base role directory
echo "Creating role directory structure for $ROLE_NAME..."
mkdir -p $ROLE_NAME

# Create standard role directories
for dir in defaults files handlers meta tasks templates tests vars
do
    mkdir -p "$ROLE_NAME/$dir"
    echo "Created directory: $dir"
done

# Create main.yml files in each directory with appropriate initial content
# defaults/main.yml - Default variables
cat > "$ROLE_NAME/defaults/main.yml" << 'EOF'
---
# defaults file for manage_server_list
netbackup_config_backup: true
netbackup_validate_only: false
netbackup_manage_server_lists: true
netbackup_ensure_repo_config: true
EOF

# handlers/main.yml - Handler definitions
cat > "$ROLE_NAME/handlers/main.yml" << 'EOF'
---
# handlers file for manage_server_list
- name: Restart NetBackup services
  service:
    name: "{{ item }}"
    state: restarted
  loop:
    - netbackup-client
  when: not netbackup_validate_only | bool
EOF

# meta/main.yml - Role metadata
cat > "$ROLE_NAME/meta/main.yml" << 'EOF'
---
galaxy_info:
  author: Your Name
  description: Manage NetBackup server lists and configuration
  company: Your Company
  license: MIT
  min_ansible_version: 2.9
  platforms:
    - name: EL
      versions:
        - 7
        - 8
    - name: Ubuntu
      versions:
        - bionic
        - focal
  galaxy_tags:
    - netbackup
    - backup
    - system
dependencies: []
EOF

# tasks/main.yml - Main tasks file
cat > "$ROLE_NAME/tasks/main.yml" << 'EOF'
---
# tasks file for manage_server_list
- name: Include OS-specific variables
  include_vars: "{{ ansible_os_family | lower }}.yml"

- name: Include validation tasks
  include_tasks: validate.yml

- name: Configure NetBackup client
  include_tasks: "{{ ansible_os_family | lower }}.yml"
  when: validation_passed | bool
EOF

# Create OS-specific task files
cat > "$ROLE_NAME/tasks/validate.yml" << 'EOF'
---
# Validation tasks
- name: Get current configuration
  command: "{{ netbackup_cmd_prefix }} {{ netbackup_bin_path }}/admincmd/nbgetconfig"
  register: current_config
  changed_when: false
EOF

# vars/linux.yml - Linux-specific variables
cat > "$ROLE_NAME/vars/linux.yml" << 'EOF'
---
# Linux-specific variables
netbackup_bin_path: /usr/openv/netbackup/bin
netbackup_conf_path: /usr/openv/netbackup/bp.conf
netbackup_cmd_prefix: "{{ lookup('env', 'SUDO_METHOD') | default('sudo', true) }}"
EOF

# templates/netbackup_config.j2 - Configuration template
cat > "$ROLE_NAME/templates/netbackup_config.j2" << 'EOF'
# NetBackup Client Configuration
# Generated by Ansible on {{ ansible_date_time.iso8601 }}
# Environment: {{ netbackup_environment }}
# Region: {{ netbackup_region }}

{% for server in netbackup_masters[netbackup_region] %}
SERVER = {{ server }}
{% endfor %}

{% for media_server in netbackup_media_servers[netbackup_region] %}
MEDIA_SERVER = {{ media_server }}
{% endfor %}

CLIENT_NAME = {{ ansible_fqdn }}
EOF

# Create a basic test playbook
mkdir -p tests
cat > "$ROLE_NAME/tests/test.yml" << 'EOF'
---
- hosts: localhost
  remote_user: root
  roles:
    - manage_server_list
EOF

# Set appropriate permissions
chmod 755 "$ROLE_NAME"
find "$ROLE_NAME" -type d -exec chmod 755 {} \;
find "$ROLE_NAME" -type f -exec chmod 644 {} \;

echo "Ansible role structure created successfully for $ROLE_NAME"
echo "Directory structure:"
tree "$ROLE_NAME"
EOF
